-- =====================================================================
-- PART 1: DATABASE & TABLE CREATION
-- =====================================================================
DROP DATABASE IF EXISTS job_portal_db;
CREATE DATABASE job_portal_db;
USE job_portal_db;

-- 1. Job seekers / candidates
CREATE TABLE job_seekers (
    seeker_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    phone VARCHAR(15),
    city VARCHAR(60),
    experience_years DECIMAL(4,1) DEFAULT 0,
    education VARCHAR(100),
    registered_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. Companies
CREATE TABLE companies (
    company_id INT AUTO_INCREMENT PRIMARY KEY,
    company_name VARCHAR(120) NOT NULL,
    industry VARCHAR(80),
    city VARCHAR(60),
    website VARCHAR(180),
    email VARCHAR(120) UNIQUE
);

-- 3. Recruiters belong to a company
CREATE TABLE recruiters (
    recruiter_id INT AUTO_INCREMENT PRIMARY KEY,
    company_id INT NOT NULL,
    recruiter_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    phone VARCHAR(15),
    FOREIGN KEY (company_id) REFERENCES companies(company_id)
);

-- 4. Job vacancies posted by companies and recruiters
CREATE TABLE jobs (
    job_id INT AUTO_INCREMENT PRIMARY KEY,
    company_id INT NOT NULL,
    recruiter_id INT NOT NULL,
    job_title VARCHAR(120) NOT NULL,
    job_description TEXT,
    job_location VARCHAR(80),
    job_type ENUM('Full-Time','Part-Time','Internship','Contract') DEFAULT 'Full-Time',
    salary_min DECIMAL(10,2),
    salary_max DECIMAL(10,2),
    experience_required DECIMAL(4,1) DEFAULT 0,
    posted_date DATE NOT NULL,
    application_deadline DATE,
    job_status ENUM('Open','Closed','On Hold') DEFAULT 'Open',
    FOREIGN KEY (company_id) REFERENCES companies(company_id),
    FOREIGN KEY (recruiter_id) REFERENCES recruiters(recruiter_id),
    CHECK (salary_min IS NULL OR salary_max IS NULL OR salary_max >= salary_min)
);

-- 5. Applications: connects job seekers and jobs
CREATE TABLE applications (
    application_id INT AUTO_INCREMENT PRIMARY KEY,
    seeker_id INT NOT NULL,
    job_id INT NOT NULL,
    applied_date DATE NOT NULL,
    application_status ENUM('Applied','Under Review','Shortlisted','Interview','Selected','Rejected') DEFAULT 'Applied',
    resume_file VARCHAR(255),
    UNIQUE (seeker_id, job_id),
    FOREIGN KEY (seeker_id) REFERENCES job_seekers(seeker_id),
    FOREIGN KEY (job_id) REFERENCES jobs(job_id)
);

-- 6. Skills catalogue
CREATE TABLE skills (
    skill_id INT AUTO_INCREMENT PRIMARY KEY,
    skill_name VARCHAR(80) NOT NULL UNIQUE
);

-- 7. Many-to-many relationship between job seekers and skills
CREATE TABLE seeker_skills (
    seeker_id INT NOT NULL,
    skill_id INT NOT NULL,
    proficiency_level ENUM('Beginner','Intermediate','Advanced') DEFAULT 'Beginner',
    PRIMARY KEY (seeker_id, skill_id),
    FOREIGN KEY (seeker_id) REFERENCES job_seekers(seeker_id),
    FOREIGN KEY (skill_id) REFERENCES skills(skill_id)
);

-- 8. Interviews scheduled for applications
CREATE TABLE interviews (
    interview_id INT AUTO_INCREMENT PRIMARY KEY,
    application_id INT NOT NULL,
    interview_date DATETIME NOT NULL,
    interview_mode ENUM('Online','In-Person','Phone') DEFAULT 'Online',
    interview_round VARCHAR(50),
    interviewer_name VARCHAR(100),
    interview_result ENUM('Pending','Passed','Failed') DEFAULT 'Pending',
    FOREIGN KEY (application_id) REFERENCES applications(application_id)
);

-- PART 4: VIEWS
-- A view is a saved SELECT query that can be used like a virtual table.
-- =====================================================================

-- 4.1 Job details with company information
CREATE OR REPLACE VIEW view_job_details AS
SELECT j.job_id,j.job_title,c.company_name,j.job_location,j.job_type,
       j.salary_min,j.salary_max,j.experience_required,j.posted_date,
       j.application_deadline,j.job_status
FROM jobs j
JOIN companies c ON j.company_id = c.company_id;

-- 4.2 Application details with candidate, job and company names
CREATE OR REPLACE VIEW view_application_details AS
SELECT a.application_id,js.full_name,js.email,j.job_title,c.company_name,
       a.applied_date,a.application_status
FROM applications a
JOIN job_seekers js ON a.seeker_id = js.seeker_id
JOIN jobs j ON a.job_id = j.job_id
JOIN companies c ON j.company_id = c.company_id;

-- 4.3 Number of applications received by each job
CREATE OR REPLACE VIEW view_job_application_count AS
SELECT j.job_id,j.job_title,c.company_name,
       COUNT(a.application_id) AS total_applications
FROM jobs j
JOIN companies c ON j.company_id = c.company_id
LEFT JOIN applications a ON j.job_id = a.job_id
GROUP BY j.job_id,j.job_title,c.company_name;

-- 4.4 Candidate skills
CREATE OR REPLACE VIEW view_candidate_skills AS
SELECT js.seeker_id,js.full_name,s.skill_name,ss.proficiency_level
FROM job_seekers js
JOIN seeker_skills ss ON js.seeker_id = ss.seeker_id
JOIN skills s ON ss.skill_id = s.skill_id;

-- 5.1 Get all applications submitted by one candidate
DROP PROCEDURE IF EXISTS sp_get_seeker_applications;
DELIMITER $$
CREATE PROCEDURE sp_get_seeker_applications(IN p_seeker_id INT)
BEGIN
    SELECT application_id,job_id,applied_date,application_status
    FROM applications
    WHERE seeker_id = p_seeker_id
    ORDER BY applied_date DESC;
END $$
DELIMITER ;

-- 5.2 Get all applicants for one job
DROP PROCEDURE IF EXISTS sp_get_job_applicants;
DELIMITER $$
CREATE PROCEDURE sp_get_job_applicants(IN p_job_id INT)
BEGIN
    SELECT a.application_id,js.full_name,js.email,js.city,
           a.applied_date,a.application_status
    FROM applications a
    JOIN job_seekers js ON a.seeker_id = js.seeker_id
    WHERE a.job_id = p_job_id
    ORDER BY a.applied_date;
END $$
DELIMITER ;

-- 5.3 Count applications by status
DROP PROCEDURE IF EXISTS sp_application_status_report;
DELIMITER $$
CREATE PROCEDURE sp_application_status_report()
BEGIN
    SELECT application_status,COUNT(*) AS total_applications
    FROM applications
    GROUP BY application_status
    ORDER BY total_applications DESC;
END $$
DELIMITER ;

-- 5.4 Update an application's status
DROP PROCEDURE IF EXISTS sp_update_application_status;
DELIMITER $$
CREATE PROCEDURE sp_update_application_status(
    IN p_application_id INT,
    IN p_new_status VARCHAR(30)
)
BEGIN
    UPDATE applications
    SET application_status = p_new_status
    WHERE application_id = p_application_id;
END $$
DELIMITER ;

-- PART 6: TRIGGERS
-- =====================================================================

-- 6.1 Record every new application in the status history table
DROP TRIGGER IF EXISTS trg_application_created;
DELIMITER $$
CREATE TRIGGER trg_application_created
AFTER INSERT ON applications
FOR EACH ROW
BEGIN
    INSERT INTO application_status_history
        (application_id,old_status,new_status)
    VALUES (NEW.application_id,NULL,NEW.application_status);
END $$
DELIMITER ;

-- 6.2 Record every application status change
DROP TRIGGER IF EXISTS trg_application_status_update;
DELIMITER $$
CREATE TRIGGER trg_application_status_update
AFTER UPDATE ON applications
FOR EACH ROW
BEGIN
    IF NOT (OLD.application_status <=> NEW.application_status) THEN
        INSERT INTO application_status_history
            (application_id,old_status,new_status)
        VALUES (NEW.application_id,OLD.application_status,NEW.application_status);
    END IF;
END $$
DELIMITER ;

-- 6.3 Prevent applications for closed jobs or after the deadline
DROP TRIGGER IF EXISTS trg_check_job_before_application;
DELIMITER $$
CREATE TRIGGER trg_check_job_before_application
BEFORE INSERT ON applications
FOR EACH ROW
BEGIN
    DECLARE v_job_status VARCHAR(20);
    DECLARE v_deadline DATE;
    SELECT job_status,application_deadline
      INTO v_job_status,v_deadline
    FROM jobs
    WHERE job_id = NEW.job_id;

    IF v_job_status <> 'Open' THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Applications are allowed only for open jobs';
    END IF;

    IF v_deadline IS NOT NULL AND NEW.applied_date > v_deadline THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'Application deadline has passed';
    END IF;
END $$
DELIMITER ;
